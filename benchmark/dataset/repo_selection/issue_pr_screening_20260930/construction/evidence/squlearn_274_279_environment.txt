    pypi-publish:
      name: Upload release to PyPI
      runs-on: ubuntu-latest
      environment:
        name: release
        url: https://pypi.org/p/squlearn
      permissions:
        id-token: write
      steps:
        - name: Set up Python
          uses: actions/setup-python@v2
          with:
            python-version: 3.9

        - name: Install dependencies
          run: pip install flit

        - uses: actions/checkout@v2
        
        - name: Build
          run: flit build

        - name: Publish package distributions to PyPI
          uses: pypa/gh-action-pypi-publish@release/v1

      - name: Install dependencies
        run: pip install black[jupyter]

      - uses: actions/checkout@v2

      - name: Run Black
        run: |
          black -l 99 src --check
          black -l 99 tests --check
          black -l 99 --ipynb examples --check

FILE .github/workflows/publish_docs.yml
name: Publish to GH Pages
on:
  release:
    types: [published]
  workflow_dispatch:

jobs:
    publish-pages:
      name: Publish to GH Pages
      runs-on: ubuntu-latest
      env:
        QISKIT_SETTINGS: ${{github.workspace}}/docs/qiskit_settings.conf
      steps:
          - name: Checkout Source Code
            uses: actions/checkout@v2
          - name: Checkout GH Pages
            uses: actions/checkout@v2
            with:
              repository: sQUlearn/squlearn.github.io
              fetch-depth: 0
              path: "docs/_build"
              token: ${{ secrets.PUSH_DOCS_TOKEN }}
          - name: Build Docs
            uses: sQUlearn/sphinx-action@master
            with:
              docs-folder: "docs/"
              pre-build-command: "pip3 install ."
              build-command: "sphinx-build -b html . _build -W"
          - name: Commit Files
            run: |
              cd docs/_build
              git config --global user.email "github-actions[bot]@users.noreply.github.com"
              git config --global user.name "github-actions[bot]"
              git add .
              git commit -m "update documentation ${{ github.event.release.tag_name }}"
          - name: Tag
            run: |
              cd docs/_build
              git tag -a "${{ github.event.release.tag_name }}" -m "${{ github.event.release.name}}"
            if: github.event_name == 'release'
          - name: Push Changes
            uses: ad-m/github-push-action@master
            with:
              github_token: ${{ secrets.PUSH_DOCS_TOKEN }}
              directory: docs/_build
              repository: sQUlearn/squlearn.github.io
              force: true
              tags: true

FILE .github/workflows/tests.yaml
name: Tests
on:
  pull_request:
  schedule:
    - cron: '23 1 * * *'

jobs:
  pytest:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.9'
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install .[examples]
      - name: Test with pytest
        run: |
          pip install pytest
          pytest tests/
      - name: Test examples
        run: |
          pip install nbmake
          pytest --nbmake --nbmake-timeout 600 examples/
